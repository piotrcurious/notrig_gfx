Yes — on ESP32 Arduino, the most practical version is usually a small fixed-point / float-hybrid transform engine:

2D: affine transforms with a 2×3 matrix

3D: quaternions or Cayley parameters for rotation

projection: homogeneous 4×4 matrix

drawing: feed the final 2D points into your display library (TFT_eSPI, LovyanGFX, U8g2, etc.)


The important part is that the engine never needs sin() / cos() in the hot path.


---

Recommended representation

1) Coordinates

Use float on ESP32 for simplicity, or int32_t fixed-point if you want tighter control.

For a practical Arduino engine, I would use:

using f32 = float;

ESP32 has hardware floating point, so this is often good enough.

2) 2D transform

Represent as:

struct Mat2D {
  f32 a, b, c, d, tx, ty; // 2x3 affine matrix
};

3) 3D rotation

Represent with a unit quaternion:

struct Quat {
  f32 w, x, y, z;
};

This avoids Euler angles and avoids trig in composition.


---

Trig-free rotation update

If you want to rotate by a small angle without sin/cos, use the half-angle parameter:

t = \tan(\Delta\theta/2)

For 2D, the incremental rotation matrix is:

R =
\frac{1}{1+t^2}
\begin{bmatrix}
1-t^2 & -2t\\
2t & 1-t^2
\end{bmatrix}

You can update orientation by multiplying by this matrix.

For 3D, use the Cayley/quaternion form. If the axis is unit length and t = tan(angle/2):

q.w = (1 - t*t) / (1 + t*t);
q.xyz = 2*t*axis / (1 + t*t);

No trig is needed if t comes from control logic, animation input, or a small-angle approximation.

Example: tiny 2D/3D engine core

This is a practical starting point.

#include <Arduino.h>
#include <math.h>

using f32 = float;

struct Vec2 {
  f32 x, y;
};

struct Vec3 {
  f32 x, y, z;
};

struct Mat2D {
  f32 a, b, c, d, tx, ty; // [a b tx; c d ty]
};

struct Quat {
  f32 w, x, y, z;
};

struct Mat4 {
  f32 m[16];
};

static inline Vec2 add(Vec2 p, Vec2 q) { return {p.x + q.x, p.y + q.y}; }
static inline Vec2 sub(Vec2 p, Vec2 q) { return {p.x - q.x, p.y - q.y}; }
static inline Vec2 scale(Vec2 p, f32 s) { return {p.x * s, p.y * s}; }

static inline Vec3 add(Vec3 p, Vec3 q) { return {p.x + q.x, p.y + q.y, p.z + q.z}; }
static inline Vec3 scale(Vec3 p, f32 s) { return {p.x * s, p.y * s, p.z * s}; }

static inline f32 dot(Vec3 a, Vec3 b) {
  return a.x*b.x + a.y*b.y + a.z*b.z;
}

static inline Vec3 cross(Vec3 a, Vec3 b) {
  return {
    a.y*b.z - a.z*b.y,
    a.z*b.x - a.x*b.z,
    a.x*b.y - a.y*b.x
  };
}

static inline f32 len(Vec3 v) {
  return sqrtf(dot(v, v));
}

static inline Vec3 normalize(Vec3 v) {
  f32 l = len(v);
  if (l <= 1e-8f) return {0,0,0};
  return scale(v, 1.0f / l);
}

static inline Quat quatNormalize(Quat q) {
  f32 l = sqrtf(q.w*q.w + q.x*q.x + q.y*q.y + q.z*q.z);
  if (l <= 1e-8f) return {1,0,0,0};
  f32 inv = 1.0f / l;
  return {q.w*inv, q.x*inv, q.y*inv, q.z*inv};
}

static inline Quat quatMul(Quat a, Quat b) {
  return {
    a.w*b.w - a.x*b.x - a.y*b.y - a.z*b.z,
    a.w*b.x + a.x*b.w + a.y*b.z - a.z*b.y,
    a.w*b.y - a.x*b.z + a.y*b.w + a.z*b.x,
    a.w*b.z + a.x*b.y - a.y*b.x + a.z*b.w
  };
}

static inline Vec3 quatRotate(Quat q, Vec3 v) {
  Vec3 u = {q.x, q.y, q.z};
  Vec3 uv = cross(u, v);
  Vec3 uuv = cross(u, uv);
  uv = scale(uv, 2.0f * q.w);
  uuv = scale(uuv, 2.0f);
  return add(v, add(uv, uuv));
}


---

2D affine transform without trig

A 2D rotation plus translation is just a matrix.

If you already have the parameter , make the matrix like this:

static inline Mat2D rot2D_from_t(f32 t, Vec2 pos) {
  f32 s = 1.0f / (1.0f + t*t);
  f32 a = (1.0f - t*t) * s;
  f32 b = (-2.0f * t) * s;
  f32 c = (2.0f * t) * s;
  f32 d = (1.0f - t*t) * s;
  return {a, b, c, d, pos.x, pos.y};
}

static inline Vec2 apply(Mat2D m, Vec2 p) {
  return {
    m.a * p.x + m.b * p.y + m.tx,
    m.c * p.x + m.d * p.y + m.ty
  };
}

That gives you a rotation matrix with no sine or cosine.

3D camera and projection

For 3D, rotate points with a quaternion and then project.

struct Camera {
  Vec3 pos;
  Quat rot;
  f32 fov_scale;   // e.g. 120.0f
  f32 near_z;      // e.g. 0.1f
};

static inline Vec2 project(Camera cam, Vec3 world, int cx, int cy) {
  Vec3 p = sub(world, cam.pos);
  // inverse camera rotation: conjugate for unit quaternion
  Quat inv = {cam.rot.w, -cam.rot.x, -cam.rot.y, -cam.rot.z};
  p = quatRotate(inv, p);

  if (p.z <= cam.near_z) return {-99999, -99999};

  f32 k = cam.fov_scale / p.z;
  return {
    cx + p.x * k,
    cy - p.y * k
  };
}

This is enough for a wireframe 3D engine.

Render loop idea

Inside loop():

1. update quaternion / transform


2. transform all vertices


3. project them


4. draw edges or filled triangles



A wireframe cube is the easiest demo.

Quat makeAxisAngleApprox(Vec3 axis, f32 t) {
  // t = tan(angle/2)
  Vec3 n = normalize(axis);
  f32 s = 1.0f / (1.0f + t*t);
  return quatNormalize({
    (1.0f - t*t) * s,
    2.0f * t * n.x * s,
    2.0f * t * n.y * s,
    2.0f * t * n.z * s
  });
}

Then each frame:

Quat spin = makeAxisAngleApprox({0,1,0}, 0.01f);
cam.rot = quatNormalize(quatMul(spin, cam.rot));

That rotates without trig.

Residual handling, practically

On ESP32, the most useful version of “residual extension” is not a formal algebraic field, but a main value + error term.

For example:

struct FExact {
  f32 v;   // main value
  f32 e;   // residual/error bound
};

Then propagate:

add errors

multiply with linearized bounds

clamp before rasterization


This is very practical if you want stable animation and collision without expensive exact arithmetic.

What I would actually build on ESP32

For a real Arduino project, I would do this:

float math

quaternion camera

2D affine sprite layer

wireframe/flat-shaded 3D mesh layer

screen-space clipping

optional residual/error tracking for collision and projection stability


That gives a usable engine with low complexity and no trig in the hot path.

Best compromise for ESP32

If you want the simplest thing that still feels mathematically clean:

use quaternions for 3D

use affine matrices for 2D

use fixed-point or float, not both unless needed

use homogeneous projection

reserve exact algebraic ideas for offline preprocessing or asset conversion
